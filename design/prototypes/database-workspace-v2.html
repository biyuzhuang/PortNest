-- 找出时长超过 120 分钟的 PG 级影片
SELECT
f.film_id,
f.title,
f.rating,
f.length,
c.name AS category
FROM film f
JOIN film_category fc ON fc.film_id = f.film_id
JOIN category c ON c.category_id = fc.category_id
WHERE f.rating = 'PG'
AND f.length > 120
ORDER BY f.title
LIMIT 100;
